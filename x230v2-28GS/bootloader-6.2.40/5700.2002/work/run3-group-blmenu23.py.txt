#!/usr/bin/env python3
# u0 (x230v2-28GS, 9600): bootloader default boot source change via Boot Menu 2 -> <opt>,
# then a plain-reload proof, then log out.
#   blmenu23.py 3   -> TFTP default (answers the TFTP prompts below), proof expects forced banner + Loading tftp
#   blmenu23.py 9   -> "Boot from default (determined by main CLI)", proof expects Loading flash:, no banner
# It confirms the option's text in "Select device" before selecting it. Bare keys in the Boot Menu,
# value+CR at the TFTP prompts, y+CR at the CLI reload confirmation. Transcript appended to LOG.
import serial, time, sys, os, re
TTY, BAUD = '/dev/u0', 9600
LOG = '/tmp/x230/run3/console-u0.log'
OPT = sys.argv[1] if len(sys.argv) > 1 else ''
if OPT not in ('3', '9'):
    print('usage: blmenu23.py 3|9'); sys.exit(2)
ANSWERS = [
    (r'Enter IP version', '4'),
    (r'Enter IP address for this device', '10.38.215.34'),
    (r'Enter subnet mask', '255.255.255.224'),
    (r'Enter gateway IP', '0.0.0.0'),
    (r'Enter TFTP server IP', '10.38.215.33'),
    (r'Enter filename', 'x230-tb470.rel'),
]
OPT_TEXT = {'3': r'3\.\s+TFTP\s+\(tftp://\)', '9': r'9\.\s+Boot from default \(determined by main CLI\)'}
os.system('stty -F {} -hupcl 2>/dev/null'.format(os.path.realpath(TTY)))
s = serial.Serial(TTY, BAUD, timeout=0.1)
t = open(LOG, 'a', errors='replace')
state = 'cli'
rc = 0
def mark(m):
    t.write('\n===== blmenu23[{}] {} {} =====\n'.format(OPT, time.strftime('%F %T'), m)); t.flush()
    print('##', time.strftime('%T'), m, flush=True)
def rd(dur):
    out = ''; end = time.time() + dur
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
    return out
def expect(pats, timeout, seed=''):
    out = seed; end = time.time() + timeout
    for p in pats:
        if re.search(p, out):
            return p, out
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
            for p in pats:
                if re.search(p, out):
                    return p, out
    return None, out
def w(b):
    s.write(b); s.flush()
def login():
    w(b'\r'); p, o = expect([r'login:\s*$', r'#\s*$', r'>\s*$', r'==>\s*$', r'new password:\s*$'], 10)
    if p and ('==>' in p or 'new password' in p or ']#' in o[-5:]):
        mark('unexpected console state {!r}: STOP'.format(o[-80:])); return False
    if p and 'login' in p:
        w(b'manager\r'); expect([r'Password:\s*$'], 10)
        w(b'friend\r'); p, o = expect([r'#\s*$', r'>\s*$', r'Login incorrect', r'new password:\s*$'], 25)
        if p and ('incorrect' in p or 'new password' in p):
            mark('login failed ({}): STOP, no further attempts'.format(p)); return False
        if p == r'>\s*$':
            w(b'enable\r'); p, o = expect([r'#\s*$'], 10)
    if p and '(config' in o[-40:]:
        w(b'end\r'); p, o = expect([r'#\s*$'], 10)
    return bool(p and '#' in p)
def cmd(c, timeout=60):
    w(c.encode() + b'\r'); p, o = expect([r'\n[^\n]*#\s*$'], timeout)
    print('=== {} ===\n{}\n=== END ==='.format(c, o), flush=True)
    return o
def reload_cli():
    global state
    w(b'reload\r'); p, o = expect([r'y/n'], 20)
    if not p:
        mark('no (y/n) prompt'); return False
    w(b'y\r'); state = 'rebooting'; return True
def boot_from_main_menu(tag):
    global state
    mark("send '9' (Quit and continue booting) [{}]".format(tag))
    w(b'9'); state = 'booting'
    p, o = expect([r'login:\s*$'], 900)
    i = o.find('Warning'); j = o.find('Booting')
    print('=== BOOT [{}] (verbatim head) ===\n{}\n=== END ==='.format(tag, o[:2000] if j < 0 else o[:j + 8]), flush=True)
    if p:
        state = 'login'; mark('reached login: prompt')
    else:
        mark('NO login: within 900 s; tail={!r}'.format(o[-300:]))
    return o
def escape_to_main():
    global state
    o = rd(1.0)
    for _ in range(4):
        tail = o[-700:]
        if 'Quit and continue booting' in tail and tail.rstrip().endswith('==>'):
            state = 'main'; return True
        if 'Return to previous menu' in tail and tail.rstrip().endswith('==>'):
            w(b'0'); o = rd(3.0); continue
        w(b'\x03'); o = rd(3.0)
    state = 'stuck'; return False
try:
    mark('open; login')
    if not login():
        rc = 2; sys.exit(rc)
    state = 'cli'
    cmd('terminal length 0'); cmd('show boot')
    mark('reload (y+CR) to reach the Boot Menu')
    if not reload_cli(): rc = 3; sys.exit(rc)
    p, o = expect([r'Ctrl\+B'], 300)
    if not p:
        mark('no bootloader banner within 300 s'); state = 'unknown'; rc = 4; sys.exit(rc)
    mark('banner seen; bare Ctrl+B until the menu appears')
    menu = ''; end = time.time() + 30
    while time.time() < end and 'Enter selection' not in menu:
        w(b'\x02'); menu += rd(0.4)
    menu += rd(1.0)
    if 'Enter selection' not in menu:
        mark('no Boot Menu'); state = 'unknown'; rc = 4; sys.exit(rc)
    state = 'main'
    print('=== BOOT MENU (verbatim) ===\n' + menu + '\n=== END ===', flush=True)
    mark("Boot Menu reached; send '2' (Change the default boot source)")
    w(b'2'); p, o = expect([r'Enter selection'], 20); o += rd(1.5); state = 'select'
    print('=== MENU 2 (verbatim) ===\n' + o + '\n=== END ===', flush=True)
    if 'Select device' not in o or not re.search(OPT_TEXT[OPT], o):
        mark('option {} text not as expected -- STOP, select nothing'.format(OPT))
        w(b'0'); escape_to_main()
        if state == 'main': boot_from_main_menu('unchanged')
        rc = 5; sys.exit(rc)
    mark("option {} confirmed; send '{}'".format(OPT, OPT))
    w(OPT.encode())
    if OPT == '3':
        state = 'prompts'; buf = ''
        for i in range(len(ANSWERS) + 2):
            p, buf = expect([r'\]:\s*$', r'==>\s*$'], 30, seed=buf)
            if not p or p.startswith('=='):
                mark('prompt not seen; escape'); escape_to_main()
                if state == 'main': boot_from_main_menu('escape')
                rc = 6; sys.exit(rc)
            line = buf.rstrip().split('\n')[-1].strip()
            ans = next((a for lab, a in ANSWERS if re.search(lab, line)), None)
            print('PROMPT {!r} -> {!r}'.format(line, ans), flush=True)
            if ans is None:
                mark('unexpected prompt {!r}; Ctrl-C and escape'.format(line)); w(b'\x03'); escape_to_main()
                if state == 'main': boot_from_main_menu('escape')
                rc = 6; sys.exit(rc)
            w(ans.encode() + b'\r'); buf = ''
            if ans == 'x230-tb470.rel':
                break
            p2, buf = expect([re.escape(ans)], 10)
            buf = buf[buf.find(ans) + len(ans):] if ans in buf else buf
    p, o = expect([r'Enter selection ==>\s*$'], 30)
    o += rd(1.0)
    print('=== AFTER SELECTION (verbatim) ===\n' + o + '\n=== END ===', flush=True)
    saved = 'Saving settings... Complete' in o
    mark('Saving settings... Complete seen: {}'.format(saved))
    if o.rfind('Select device') > o.rfind('Boot Menu:'):
        mark("still at Select device; send '0'"); w(b'0'); expect([r'Quit and continue booting'], 10); rd(1.0)
    state = 'main'
    boot_from_main_menu('after 2->{}'.format(OPT))
    if state != 'login': rc = 7; sys.exit(rc)
    if not saved: rc = 8; sys.exit(rc)
    mark('PROOF: plain CLI reload, no keypresses in the bootloader')
    if not login(): rc = 9; sys.exit(rc)
    state = 'cli'
    if not reload_cli(): rc = 9; sys.exit(rc)
    p, o = expect([r'login:\s*$'], 900)
    state = 'login' if p else 'unknown'
    forced = 'non-standard location' in o
    src = re.search(r'(Loading [^\r\n]*|Reading [^\r\n]*)', o)
    i = o.find('Bootloader'); j = o.find('Booting')
    print('=== PROOF BOOT (bootloader part, verbatim) ===\n' + (o[i:j + 8] if i >= 0 and j > i else o[-2500:]) + '\n=== END ===', flush=True)
    mark('proof boot: login={} forced_banner={} source={!r}'.format(bool(p), forced, src.group(1) if src else None))
    want_tftp = (OPT == '3')
    ok = bool(p) and (forced == want_tftp) and bool(src) and (('tftp://10.38.215.33/x230-tb470.rel' in src.group(1)) if want_tftp else src.group(1).startswith('Loading flash:'))
    if not ok: rc = 10
    if p:
        if login():
            state = 'cli'
            cmd('terminal length 0'); cmd('show system'); cmd('show boot')
            mark('exit (log out)')
            w(b'exit\r'); p, o = expect([r'login:\s*$'], 15)
            if p: state = 'loggedout'
finally:
    if state == 'main':
        try: boot_from_main_menu('finally')
        except Exception as e: mark('finally boot failed: {}'.format(e))
    mark('close (state={}, rc={})'.format(state, rc))
    t.close(); s.close()
    sys.exit(rc)
