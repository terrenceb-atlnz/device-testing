#!/usr/bin/env python3
# Recover u0 after the stopped 5700 run: flash holds no release and the unit boot-loops
# ("There is no primary release file set ... Boot failed ... Restarting").
# Phase A: catch "Press <Ctrl+B>", Boot Menu 1 (one-off, saves nothing) -> 3 TFTP -> x230-tb470.rel -> login.
# Phase B: inspect; temporary vlan1 IP; copy the bench release back to flash; boot system; clean the
#          suite's leftover .cfg/.rel; boot config default.cfg; reload to flash.
# Phase C: verify the flash boot (no forced banner), show boot / running-config; exit.
# Stops on any unexpected '% ' line in phase B (prints it) and leaves the unit at a prompt.
import serial, time, sys, os, re
TTY, BAUD = '/dev/u0', 9600
LOG = '/tmp/x230/console-u0.log'
ANSWERS = [
    (r'Enter IP version', '4'),
    (r'Enter IP address for this device', '10.38.215.34'),
    (r'Enter subnet mask', '255.255.255.224'),
    (r'Enter gateway IP', '0.0.0.0'),
    (r'Enter TFTP server IP', '10.38.215.33'),
    (r'Enter filename', 'x230-tb470.rel'),
]
os.system('stty -F {} -hupcl 2>/dev/null'.format(os.path.realpath(TTY)))
s = serial.Serial(TTY, BAUD, timeout=0.1)
t = open(LOG, 'a', errors='replace')
state = 'unknown'
def mark(m):
    t.write('\n===== recover1 {} {} =====\n'.format(time.strftime('%F %T'), m)); t.flush()
    print('##', time.strftime('%T'), m, flush=True)
def rd(dur):
    out = ''; end = time.time() + dur
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
    return out
def expect(pats, timeout, seed=''):
    out = seed; end = time.time() + timeout
    for p in pats:
        if re.search(p, out):
            return p, out
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
            for p in pats:
                if re.search(p, out):
                    return p, out
    return None, out
def w(b):
    s.write(b); s.flush()
def login():
    w(b'\r'); p, o = expect([r'login:\s*$', r'#\s*$', r'>\s*$'], 10)
    if p and 'login' in p:
        w(b'manager\r'); expect([r'Password:\s*$'], 10)
        w(b'friend\r'); p, o = expect([r'#\s*$', r'>\s*$', r'Login incorrect'], 25)
        if p == r'>\s*$':
            w(b'enable\r'); p, o = expect([r'#\s*$'], 10)
    return bool(p and '#' in p)
def cmd(c, timeout=60, allow_err=False):
    w(c.encode() + b'\r'); p, o = expect([r'\n[^\n]*#\s*$'], timeout)
    print('=== {} ===\n{}\n=== END ==='.format(c, o), flush=True)
    if p is None:
        mark('no prompt after {!r}'.format(c)); raise SystemExit(20)
    if re.search(r'^\s*% ', o, re.M) and not allow_err:
        mark('CLI error after {!r}: STOP'.format(c)); raise SystemExit(21)
    return o
def escape_to_flash(tag):
    global state
    mark('ESCAPE [{}]: Ctrl-C / 0 / 9'.format(tag))
    o = rd(1.0)
    for _ in range(4):
        tail = o[-600:]
        if 'Quit and continue booting' in tail and tail.rstrip().endswith('==>'):
            w(b'9'); state = 'booting'; return
        if 'Return to previous menu' in tail and tail.rstrip().endswith('==>'):
            w(b'0'); o = rd(3.0); continue
        w(b'\x03'); o = rd(3.0)
    state = 'stuck'

rc = 0
try:
    # ---------- Phase A ----------
    mark('phase A: wait for the bootloader banner (unit is restart-looping)')
    w(b'\r')
    p, o = expect([r'Ctrl\+B', r'login:\s*$', r'#\s*$'], 120)
    if p is None:
        mark('no banner / prompt within 120 s; tail={!r}'.format(o[-300:])); sys.exit(3)
    if 'Ctrl' not in p:
        mark('unit is at a CLI prompt, not boot-looping ({!r}); STOP, nothing sent'.format(p)); sys.exit(4)
    menu = ''; end = time.time() + 30
    while time.time() < end and 'Enter selection' not in menu:
        w(b'\x02'); menu += rd(0.4)
    menu += rd(1.0)
    if 'Enter selection' not in menu:
        mark('no Boot Menu'); sys.exit(3)
    state = 'main'
    print('=== BOOT MENU (verbatim) ===\n' + menu[-1500:] + '\n=== END ===', flush=True)
    if '1. Perform one-off boot from alternate source' not in re.sub(r'[ \t]+', ' ', menu):
        escape_to_flash('menu mismatch'); sys.exit(4)
    w(b'1'); p, o = expect([r'Enter selection'], 20); o += rd(1.0); state = 'select'
    if not re.search(r'3\.\s+TFTP', o):
        escape_to_flash('no TFTP option'); sys.exit(4)
    w(b'3'); state = 'prompts'; buf = ''
    for i in range(len(ANSWERS) + 2):
        p, buf = expect([r'\]:\s*$', r'==>\s*$'], 30, seed=buf)
        if not p or p.startswith('=='):
            escape_to_flash('prompt'); sys.exit(5)
        line = buf.rstrip().split('\n')[-1].strip()
        ans = next((a for lab, a in ANSWERS if re.search(lab, line)), None)
        print('PROMPT {!r} -> {!r}'.format(line, ans), flush=True)
        if ans is None:
            escape_to_flash('unexpected prompt'); sys.exit(5)
        w(ans.encode() + b'\r'); buf = ''
        if ans == 'x230-tb470.rel':
            break
        p2, buf = expect([re.escape(ans)], 10)
        buf = buf[buf.find(ans) + len(ans):] if ans in buf else buf
    state = 'tftp'
    p, o = expect([r'login:\s*$', r'Enter selection ==>\s*$'], 900)
    i = o.find('x230-tb470.rel'); j = o.find('Booting')
    print('=== TFTP BOOT (verbatim) ===\n' + (o[max(i, 0):j + 8] if j > 0 else o[-2000:]) + '\n=== END ===', flush=True)
    if not p or 'Enter' in p:
        mark('TFTP boot did not reach login'); state = 'main' if p else 'unknown'; rc = 10; sys.exit(rc)
    state = 'login'
    # ---------- Phase B ----------
    mark('phase B: CLI')
    if not login(): rc = 11; sys.exit(rc)
    state = 'cli'
    cmd('terminal length 0')
    cmd('show system'); cmd('show boot'); cmd('dir', allow_err=True); cmd('show running-config')
    cmd('show interface brief', allow_err=True)
    cmd('configure terminal'); cmd('interface vlan1'); cmd('ip address 10.38.215.74/27'); cmd('end')
    rd(5.0)
    cmd('ping 10.38.215.65 repeat 3', timeout=30, allow_err=True)
    o = cmd('copy tftp://10.38.215.65/x230v2_28GS-tb470.rel flash:/x230v2_28GS-tb470.rel', timeout=600, allow_err=True)
    if 'Successful operation' not in o:
        mark('copy did not report Successful operation: STOP (unit stays on the TFTP-booted image)'); rc = 12; sys.exit(rc)
    cmd('dir *.rel')
    cmd('configure terminal')
    cmd('boot system flash:/x230v2_28GS-tb470.rel')
    cmd('no boot system backup', allow_err=True)
    cmd('boot config-file flash:/default.cfg', allow_err=True)
    cmd('end')
    o = cmd('dir *.cfg', allow_err=True)
    for f in re.findall(r'flash:/(\S+\.cfg)', o):
        if f.startswith('swi_a_5700') or f.startswith('TestCase') or f == 'default.cfg':
            cmd('delete force flash:/{}'.format(f), allow_err=True)
    cmd('dir', allow_err=True)
    cmd('show boot')
    mark('reload to flash')
    w(b'reload\r')
    p, o = expect([r'y/n'], 20)
    if not p: rc = 13; sys.exit(rc)
    if 'save' in o.lower().split('\n')[-1]:
        mark('save prompt {!r}: answer n'.format(o[-200:])); w(b'n\r'); p, o = expect([r'reboot system\? \(y/n\)'], 20)
    w(b'y\r'); state = 'rebooting'
    # ---------- Phase C ----------
    p, o = expect([r'login:\s*$', r'Enter selection ==>\s*$', r'Boot failed'], 900)
    forced = 'non-standard location' in o
    i = o.find('Bootloader'); j = o.find('Booting')
    print('=== FLASH BOOT (verbatim) ===\n' + (o[i:j + 8] if i >= 0 and j > i else o[-2000:]) + '\n=== END ===', flush=True)
    mark('flash boot: p={!r} forced_banner={}'.format(p, forced))
    if p is None or 'login' not in p: rc = 14; state = 'unknown'; sys.exit(rc)
    state = 'login'
    if not login(): rc = 15; sys.exit(rc)
    state = 'cli'
    cmd('terminal length 0')
    cmd('show system'); cmd('show boot'); cmd('dir', allow_err=True); cmd('show running-config'); cmd('show license brief', allow_err=True)
    mark('phase C done; exit (log out)')
    w(b'exit\r'); expect([r'login:\s*$'], 15); state = 'loggedout'
finally:
    if state in ('main', 'select', 'prompts'):
        try: escape_to_flash('finally')
        except Exception as e: mark('finally escape failed: {}'.format(e))
    mark('close (state={}, rc={})'.format(state, rc))
    t.close(); s.close()
    open('/tmp/x230/run1/recover1.done', 'w').write('state={} rc={}\n'.format(state, rc))
