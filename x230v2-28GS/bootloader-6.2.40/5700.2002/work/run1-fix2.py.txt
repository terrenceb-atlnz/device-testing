#!/usr/bin/env python3
# u0 after recover1: factory defaults (default.cfg deleted) -> sitting at the forced
# "% Default password needs to be changed. / Enter new password:" dialog.
# Phase 1: answer the dialog with the framework's credential (friend) so manager/friend is explicit
#          in config like the pre-run baseline; refuse anything unexpected (Ctrl-C, stop).
#          Then terminal length 0, show running-config, copy running-config flash:/default.cfg,
#          show boot, dir.
# Phase 2 (Test Engineer, relayed ~15:03): Boot Menu 2 -> 3 TFTP default: IPv4, 10.38.215.34,
#          255.255.255.224, 0.0.0.0, 10.38.215.33, x230-tb470.rel; expect "Saving settings... Complete";
#          then main menu 9 (continue booting) -> boots per the new default.
# Phase 3: one plain CLI reload as proof (TFTP boot + forced banner expected) -> login, show system/boot; exit.
import serial, time, sys, os, re
TTY, BAUD = '/dev/u0', 9600
LOG = '/tmp/x230/console-u0.log'
PW = 'friend'
PW_TMP = 'P@ssw0rd'   # framework password list member; replaced by friend in config below
ANSWERS = [
    (r'Enter IP version', '4'),
    (r'Enter IP address for this device', '10.38.215.34'),
    (r'Enter subnet mask', '255.255.255.224'),
    (r'Enter gateway IP', '0.0.0.0'),
    (r'Enter TFTP server IP', '10.38.215.33'),
    (r'Enter filename', 'x230-tb470.rel'),
]
os.system('stty -F {} -hupcl 2>/dev/null'.format(os.path.realpath(TTY)))
s = serial.Serial(TTY, BAUD, timeout=0.1)
t = open(LOG, 'a', errors='replace')
state = 'dialog'
def mark(m):
    t.write('\n===== fix2 {} {} =====\n'.format(time.strftime('%F %T'), m)); t.flush()
    print('##', time.strftime('%T'), m, flush=True)
def rd(dur):
    out = ''; end = time.time() + dur
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
    return out
def expect(pats, timeout, seed=''):
    out = seed; end = time.time() + timeout
    for p in pats:
        if re.search(p, out):
            return p, out
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
            for p in pats:
                if re.search(p, out):
                    return p, out
    return None, out
def w(b):
    s.write(b); s.flush()
def cmd(c, timeout=60, allow_err=False):
    w(c.encode() + b'\r'); p, o = expect([r'\n[^\n]*#\s*$'], timeout)
    print('=== {} ===\n{}\n=== END ==='.format(c, o), flush=True)
    if p is None:
        mark('no prompt after {!r}'.format(c)); raise SystemExit(20)
    if re.search(r'^\s*% ', o, re.M) and not allow_err:
        mark('CLI error after {!r}: STOP'.format(c)); raise SystemExit(21)
    return o
def login():
    w(b'\r'); p, o = expect([r'login:\s*$', r'#\s*$', r'>\s*$', r'new password:\s*$'], 10)
    if p and 'new password' in p:
        mark('forced password dialog at login: STOP'); return False
    if p and 'login' in p:
        w(b'manager\r'); expect([r'Password:\s*$'], 10)
        w(PW.encode() + b'\r'); p, o = expect([r'#\s*$', r'>\s*$', r'Login incorrect', r'new password:\s*$'], 25)
        if p and 'new password' in p:
            mark('forced password dialog after login: STOP'); w(b'\x03'); return False
        if p == r'>\s*$':
            w(b'enable\r'); p, o = expect([r'#\s*$'], 10)
    return bool(p and '#' in p)
def escape_to_boot(tag):
    global state
    mark('ESCAPE [{}]'.format(tag)); o = rd(1.0)
    for _ in range(4):
        tail = o[-600:]
        if 'Quit and continue booting' in tail and tail.rstrip().endswith('==>'):
            w(b'9'); state = 'booting'; return
        if 'Return to previous menu' in tail and tail.rstrip().endswith('==>'):
            w(b'0'); o = rd(3.0); continue
        w(b'\x03'); o = rd(3.0)
    state = 'stuck'

rc = 0
try:
    # ---------- Phase 1 ----------
    mark('phase 1: read the console without sending')
    o = rd(2.0)
    tail = open(LOG, errors='replace').read()[-1500:]
    tail = re.sub(r'\n===== [^\n]*=====\n?', '\n', tail)
    print('=== CONSOLE STATE (transcript tail, nothing sent) ===\n{}\n=== END ==='.format(tail[-400:] + o), flush=True)
    p = r'new password:\s*$' if re.search(r'Enter new password:\s*$', (tail + o).rstrip() + ' ') else None
    if p is None:
        w(b'\r'); p, o2 = expect([r'login:\s*$', r'#\s*$', r'>\s*$', r'Password:\s*$', r'new password:\s*$'], 10)
        print('=== AFTER CR ===\n{}\n=== END ==='.format(o2[-400:]), flush=True)
    if p and 'new password' in p:
        mark('at "Enter new password:" (a CR there re-prompts); answer with the framework credential')
        w(PW_TMP.encode() + b'\r')
        p, o = expect([r'Re-enter new password:\s*$', r'[Cc]onfirm[^\n]*:\s*$', r'[Rr]etype[^\n]*:\s*$', r'Enter new password:\s*$', r'#\s*$', r'>\s*$', r'%[^\n]*'], 15)
        print('=== AFTER NEW PASSWORD ===\n{}\n=== END ==='.format(o[-600:]), flush=True)
        if p and ('onfirm' in p or 'etype' in p or 'Re-enter' in p):
            w(PW_TMP.encode() + b'\r')
            p, o = expect([r'#\s*$', r'>\s*$', r'new password:\s*$', r'login:\s*$'], 20)
            print('=== AFTER CONFIRM ===\n{}\n=== END ==='.format(o[-600:]), flush=True)
        if not p or not ('#' in p or '>' in p):
            mark('dialog did not end at a CLI prompt (p={!r}): Ctrl-C and STOP'.format(p))
            w(b'\x03'); rd(2.0); rc = 2; sys.exit(rc)
        if '>' in p:
            w(b'enable\r'); expect([r'#\s*$'], 10)
    elif p and ('login' in p or 'Password' in p):
        if 'Password' in p:
            w(b'\x03'); rd(2.0)
        if not login():
            rc = 3; sys.exit(rc)
    elif not p:
        mark('no recognisable prompt: STOP'); rc = 4; sys.exit(rc)
    state = 'cli'
    cmd('terminal length 0')
    cmd('configure terminal')
    cmd('username manager privilege 15 password friend')
    cmd('end')
    cmd('show running-config')
    cmd('copy running-config flash:/default.cfg', timeout=60)
    cmd('show boot'); cmd('dir', allow_err=True); cmd('show system')
    # verify login with manager/friend works in a fresh session (proves no forced change next time)
    mark('log out and back in to prove manager/friend with no forced change')
    w(b'exit\r'); expect([r'login:\s*$'], 15)
    if not login():
        rc = 5; sys.exit(rc)
    cmd('terminal length 0')
    # ---------- Phase 2 ----------
    mark('phase 2: reload to the Boot Menu, 2 -> 3 (TFTP default)')
    w(b'reload\r'); p, o = expect([r'y/n'], 20)
    if not p: rc = 6; sys.exit(rc)
    w(b'y\r'); state = 'rebooting'
    p, o = expect([r'Ctrl\+B'], 300)
    if not p: rc = 7; sys.exit(rc)
    menu = ''; end = time.time() + 30
    while time.time() < end and 'Enter selection' not in menu:
        w(b'\x02'); menu += rd(0.4)
    menu += rd(1.0)
    if 'Enter selection' not in menu: rc = 7; sys.exit(rc)
    state = 'main'
    w(b'2'); p, o = expect([r'Enter selection'], 20); o += rd(1.5); state = 'select'
    print('=== MENU 2 (verbatim) ===\n' + o + '\n=== END ===', flush=True)
    if 'Select device' not in o or not re.search(r'3\.\s+TFTP', o):
        escape_to_boot('menu 2 mismatch'); rc = 8; sys.exit(rc)
    w(b'3'); state = 'prompts'; buf = ''
    answered = []
    for i in range(len(ANSWERS) + 2):
        p, buf = expect([r'\]:\s*$', r'==>\s*$'], 30, seed=buf)
        if not p or p.startswith('=='):
            escape_to_boot('prompt'); rc = 9; sys.exit(rc)
        line = buf.rstrip().split('\n')[-1].strip()
        ans = next((a for lab, a in ANSWERS if re.search(lab, line)), None)
        print('PROMPT {!r} -> {!r}'.format(line, ans), flush=True)
        if ans is None:
            escape_to_boot('unexpected prompt'); rc = 9; sys.exit(rc)
        w(ans.encode() + b'\r'); answered.append((line, ans)); buf = ''
        if ans == 'x230-tb470.rel':
            break
        p2, buf = expect([re.escape(ans)], 10)
        buf = buf[buf.find(ans) + len(ans):] if ans in buf else buf
    p, o = expect([r'Enter selection ==>\s*$'], 30)
    o += rd(1.0)
    print('=== AFTER FILENAME (verbatim) ===\n' + o + '\n=== END ===', flush=True)
    saved = 'Saving settings... Complete' in o
    mark('Saving settings... Complete seen: {}'.format(saved))
    tail = o[-600:]
    if 'Return to previous menu' in tail:
        w(b'0'); o = rd(3.0); tail = o[-600:]
    state = 'main'
    mark("main menu: '9' (Quit and continue booting) -> boots per the new default")
    w(b'9'); state = 'booting'
    p, o = expect([r'login:\s*$', r'Enter selection ==>\s*$'], 900)
    i = o.find('Bootloader'); j = o.find('Booting')
    print('=== BOOT AFTER 2->3 (verbatim, from menu exit) ===\n' + o[:3000 if j < 0 else j + 8] + '\n=== END ===', flush=True)
    if not p or 'login' not in p: rc = 10; state = 'unknown'; sys.exit(rc)
    state = 'login'
    if not saved: rc = 11; sys.exit(rc)
    # ---------- Phase 3 ----------
    if not login(): rc = 12; sys.exit(rc)
    state = 'cli'
    cmd('terminal length 0'); cmd('show system')
    mark('phase 3: plain CLI reload as proof (no keys in the bootloader)')
    w(b'reload\r'); p, o = expect([r'y/n'], 20)
    if not p: rc = 13; sys.exit(rc)
    w(b'y\r'); state = 'rebooting'
    p, o = expect([r'login:\s*$'], 900)
    forced = 'non-standard location' in o
    i = o.find('Bootloader'); j = o.find('Booting')
    print('=== PROOF BOOT (verbatim) ===\n' + (o[i:j + 8] if i >= 0 and j > i else o[-2500:]) + '\n=== END ===', flush=True)
    mark('proof boot: login={} forced_banner={} tftp={}'.format(bool(p), forced, 'Loading tftp' in o))
    if not p: rc = 14; state = 'unknown'; sys.exit(rc)
    state = 'login'
    if not login(): rc = 15; sys.exit(rc)
    state = 'cli'
    cmd('terminal length 0'); cmd('show system'); cmd('show boot'); cmd('dir', allow_err=True)
    cmd('show running-config'); cmd('show license brief', allow_err=True)
    mark('exit (log out)')
    w(b'exit\r'); expect([r'login:\s*$'], 15); state = 'loggedout'
finally:
    if state in ('main', 'select', 'prompts'):
        try: escape_to_boot('finally')
        except Exception as e: mark('finally escape failed: {}'.format(e))
    mark('close (state={}, rc={})'.format(state, rc))
    t.close(); s.close()
    open('/tmp/x230/run1/fix2.done', 'w').write('state={} rc={}\n'.format(state, rc))
