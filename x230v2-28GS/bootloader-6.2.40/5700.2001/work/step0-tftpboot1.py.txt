#!/usr/bin/env python3
# Step 0 (Test Engineer: "Prove TFTP, then launch"): ONE one-off TFTP boot of u0 via the USB
# Ethernet adapter (Boot Menu 1 -> Select device 3), answer each prompt by its LABEL, confirm
# login + show system, then a plain reload back to flash, then log out.
# Nothing is saved in the bootloader (menu 1 is one-off). Any unexpected prompt -> abort to flash.
import serial, time, sys, os, re, threading
TTY, BAUD = '/dev/u0', 9600
LOG = '/tmp/x230/console-u0.log'
TRACE = '/tmp/x230/run1/eth2.trace'
ANSWERS = [  # (label regex, answer)
    (r'Enter IP version', '4'),
    (r'Enter IP address for this device', '10.38.215.34'),
    (r'Enter subnet mask', '255.255.255.224'),
    (r'Enter gateway IP', '0.0.0.0'),
    (r'Enter TFTP server IP', '10.38.215.33'),
    (r'Enter filename', 'x230-tb470.rel'),
]
os.system('stty -F {} -hupcl 2>/dev/null'.format(os.path.realpath(TTY)))
s = serial.Serial(TTY, BAUD, timeout=0.1)
t = open(LOG, 'a', errors='replace')
state = 'cli'
stop_trace = False

def tracer():
    with open(TRACE, 'a') as f:
        f.write('# time carrier operstate tx_bytes rx_bytes\n')
        while not stop_trace:
            vals = []
            for n in ('carrier', 'operstate', 'statistics/tx_bytes', 'statistics/rx_bytes'):
                try:
                    vals.append(open('/sys/class/net/eth2/' + n).read().strip())
                except Exception:
                    vals.append('?')
            f.write('{} {}\n'.format(time.strftime('%T'), ' '.join(vals))); f.flush()
            time.sleep(1)

def mark(m):
    t.write('\n===== tftpboot1 {} {} =====\n'.format(time.strftime('%F %T'), m)); t.flush()
    print('##', time.strftime('%T'), m, flush=True)
def rd(dur):
    out = ''; end = time.time() + dur
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
    return out
def expect(pats, timeout, seed=''):
    out = seed; end = time.time() + timeout
    for p in pats:
        if re.search(p, out):
            return p, out
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
            for p in pats:
                if re.search(p, out):
                    return p, out
    return None, out
def w(b):
    s.write(b); s.flush()
def login():
    w(b'\r'); p, o = expect([r'login:\s*$', r'#\s*$', r'>\s*$'], 10)
    if p and 'login' in p:
        w(b'manager\r'); expect([r'Password:\s*$'], 10)
        w(b'friend\r'); p, o = expect([r'#\s*$', r'>\s*$', r'Login incorrect'], 25)
        if p == r'>\s*$':
            w(b'enable\r'); p, o = expect([r'#\s*$'], 10)
    return bool(p and '#' in p)
def cmd(c, timeout=30):
    w(c.encode() + b'\r'); p, o = expect([r'\n[^\n]*#\s*$'], timeout)
    print('=== {} ===\n{}\n=== END ==='.format(c, o), flush=True)
    return o
def reload_cli():
    global state
    w(b'reload\r'); p, o = expect([r'y/n'], 20)
    if not p:
        mark('no (y/n) prompt'); return False
    w(b'y\r'); state = 'rebooting'; return True
def escape_to_flash(tag):
    """From wherever the bootloader is: Ctrl-C out of a prompt, 0 out of Select device, 9 from main."""
    global state
    mark('ESCAPE to flash boot [{}]'.format(tag))
    o = rd(1.0)
    for _ in range(4):
        tail = o[-600:]
        if 'Quit and continue booting' in tail and tail.rstrip().endswith('==>'):
            mark("main menu: send '9' (Quit and continue booting)")
            w(b'9'); state = 'booting'
            p, o2 = expect([r'login:\s*$'], 900)
            state = 'login' if p else 'unknown'
            mark('escape boot: login={}'.format(bool(p)))
            return o2
        if 'Return to previous menu' in tail and tail.rstrip().endswith('==>'):
            mark("Select device: send '0'"); w(b'0'); o = rd(3.0); continue
        mark('in a prompt or unknown: Ctrl-C'); w(b'\x03'); o = rd(3.0)
        if not o.strip():
            w(b'\r'); o = rd(2.0)
    mark('could not reach a known menu; tail={!r}'.format(o[-300:])); state = 'stuck'
    return o

th = threading.Thread(target=tracer, daemon=True); th.start()
rc = 0
try:
    mark('open; login')
    if not login():
        mark('no privileged prompt; abort'); sys.exit(2)
    w(b'terminal length 0\r'); expect([r'#\s*$'], 10)
    cmd('show system')
    cmd('show boot')
    mark('reload (y+CR) to reach the Boot Menu')
    if not reload_cli(): sys.exit(2)
    p, o = expect([r'Ctrl\+B'], 300)
    if not p:
        mark('no bootloader banner within 300 s'); state = 'unknown'; sys.exit(3)
    mark('banner seen; bare Ctrl+B until the menu appears')
    menu = ''; end = time.time() + 30
    while time.time() < end and 'Enter selection' not in menu:
        w(b'\x02'); menu += rd(0.4)
    menu += rd(1.0)
    if 'Enter selection' not in menu:
        mark('no Boot Menu'); state = 'unknown'; sys.exit(3)
    state = 'main'
    if '1. Perform one-off boot from alternate source' not in re.sub(r'[ \t]+', ' ', menu):
        mark('main-menu option 1 is not one-off boot -- STOP'); escape_to_flash('menu1 mismatch'); sys.exit(4)
    mark("Boot Menu reached; send '1' (Perform one-off boot from alternate source)")
    w(b'1'); p, o = expect([r'Enter selection'], 20); o += rd(1.0)
    print('=== MENU 1 (verbatim) ===\n' + o + '\n=== END ===', flush=True)
    state = 'select'
    if 'Select device' not in o or not re.search(r'3\.\s+TFTP', o):
        mark("option 3 is not TFTP -- STOP"); escape_to_flash('menu1->3 mismatch'); sys.exit(4)
    mark("send '3' (TFTP)")
    w(b'3'); state = 'prompts'
    buf = ''
    answered = []
    for i in range(len(ANSWERS) + 2):
        p, buf = expect([r'\]:\s*$', r'==>\s*$'], 30, seed=buf)
        if not p:
            mark('no prompt within 30 s; tail={!r}'.format(buf[-300:])); escape_to_flash('no prompt'); sys.exit(5)
        if p.startswith('==>'):
            mark('back at a menu unexpectedly; tail={!r}'.format(buf[-400:])); escape_to_flash('menu instead of prompt'); sys.exit(5)
        # the prompt line is the last line of buf
        line = buf.rstrip().split('\n')[-1].strip()
        ans = None
        for lab, a in ANSWERS:
            if re.search(lab, line):
                ans = a; break
        print('PROMPT {!r} -> {!r}'.format(line, ans), flush=True)
        if ans is None:
            mark('UNEXPECTED prompt {!r}: abort'.format(line)); escape_to_flash('unexpected prompt'); sys.exit(5)
        w(ans.encode() + b'\r'); answered.append((line, ans)); buf = ''
        if ans == 'x230-tb470.rel':
            break
        # consume the echo so the next search starts after it
        p2, buf = expect([re.escape(ans)], 10)
        buf = buf[buf.find(ans) + len(ans):] if ans in buf else buf
    print('=== PROMPTS ANSWERED ===\n' + '\n'.join('{!r} -> {!r}'.format(a, b) for a, b in answered) + '\n=== END ===', flush=True)
    state = 'tftp'
    mark('filename sent; waiting for download / verify / login (900 s)')
    p, o = expect([r'login:\s*$', r'Enter selection ==>\s*$'], 900)
    i = o.find('x230-tb470.rel'); j = o.find('Booting')
    print('=== TFTP BOOT (bootloader part, verbatim) ===\n' + (o[max(i, 0):j + 8] if j > 0 else o[-3000:]) + '\n=== END ===', flush=True)
    if p is None or 'Enter selection' in p:
        mark('TFTP boot did NOT reach login (p={!r}); tail={!r}'.format(p, o[-500:]))
        state = 'main' if p else 'unknown'
        escape_to_flash('tftp failed'); rc = 10; sys.exit(rc)
    state = 'login'
    mark('TFTP boot reached login:; verify')
    if not login():
        mark('no privileged prompt after TFTP boot'); rc = 11; sys.exit(rc)
    w(b'terminal length 0\r'); expect([r'#\s*$'], 10)
    cmd('show system'); cmd('show boot')
    mark('plain CLI reload back to flash (no keypresses in the bootloader)')
    if not reload_cli(): rc = 12; sys.exit(rc)
    p, o = expect([r'login:\s*$'], 900)
    state = 'login' if p else 'unknown'
    forced = 'non-standard location' in o
    rdline = re.search(r'(Reading [^\r\n]*|Loading [^\r\n]*)', o)
    mark('flash boot: login={} forced_banner={} source={!r}'.format(bool(p), forced, rdline.group(1) if rdline else None))
    i = o.find('Bootloader'); j = o.find('Booting')
    print('=== FLASH BOOT (bootloader part, verbatim) ===\n' + (o[i:j + 8] if i >= 0 and j > i else o[:1500]) + '\n=== END ===', flush=True)
    if not p: rc = 13; sys.exit(rc)
    if not login(): rc = 14; sys.exit(rc)
    w(b'terminal length 0\r'); expect([r'#\s*$'], 10)
    cmd('show system'); cmd('show boot')
    mark('log out (exit)')
    w(b'exit\r'); expect([r'login:\s*$'], 15)
    state = 'loggedout'
finally:
    if state in ('main', 'select', 'prompts'):
        try: escape_to_flash('finally')
        except Exception as e: mark('finally escape failed: {}'.format(e))
    stop_trace = True; time.sleep(1.2)
    mark('close (state={})'.format(state))
    t.close(); s.close()
    open('/tmp/x230/run1/tftpboot1.done', 'w').write('state={}\n'.format(state))
