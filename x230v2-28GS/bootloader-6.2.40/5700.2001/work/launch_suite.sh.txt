#!/bin/bash
# Detached wrapper for the 5700 suite run (row #1). Runs as root via sudo -n setsid nohup.
# Writes start/end stamps, the exit code and a done-marker under /tmp/x230/run1/.
RUN=/home/terrenceb/claude/device-testing/x230v2-28GS/bootloader-6.2.40/5700_x230v2-28GS_6.2.40
M=/tmp/x230/run1
cd "$RUN" || { echo "cd failed" > $M/suite.rc; touch $M/suite.done; exit 1; }
echo "start $(date '+%F %T %Z') pid $$" > $M/suite.start
./runTestSuite.py -s default.setup -u -v > runTestSuite.stdout 2>&1 < /dev/null
rc=$?
echo "$rc" > $M/suite.rc
echo "end $(date '+%F %T %Z') rc $rc" >> $M/suite.start
touch $M/suite.done
