#!/usr/bin/env python3
# u0 after recover1: factory defaults (default.cfg deleted) -> sitting at the forced
# "% Default password needs to be changed. / Enter new password:" dialog.
# Phase 1: answer the dialog with the framework's credential (friend) so manager/friend is explicit
#          in config like the pre-run baseline; refuse anything unexpected (Ctrl-C, stop).
#          Then terminal length 0, show running-config, copy running-config flash:/default.cfg,
#          show boot, dir.
# Phase 2 (Test Engineer, relayed ~15:03): Boot Menu 2 -> 3 TFTP default: IPv4, 10.38.215.34,
#          255.255.255.224, 0.0.0.0, 10.38.215.33, x230-tb470.rel; expect "Saving settings... Complete";
#          then main menu 9 (continue booting) -> boots per the new default.
# Phase 3: one plain CLI reload as proof (TFTP boot + forced banner expected) -> login, show system/boot; exit.
import serial, time, sys, os, re
TTY, BAUD = '/dev/u0', 9600
LOG = '/tmp/x230/console-u0.log'
PW = 'friend'
PW_TMP = 'P@ssw0rd'   # framework password list member; replaced by friend in config below
ANSWERS = [
    (r'Enter IP version', '4'),
    (r'Enter IP address for this device', '10.38.215.34'),
    (r'Enter subnet mask', '255.255.255.224'),
    (r'Enter gateway IP', '0.0.0.0'),
    (r'Enter TFTP server IP', '10.38.215.33'),
    (r'Enter filename', 'x230-tb470.rel'),
]
os.system('stty -F {} -hupcl 2>/dev/null'.format(os.path.realpath(TTY)))
s = serial.Serial(TTY, BAUD, timeout=0.1)
t = open(LOG, 'a', errors='replace')
state = 'dialog'
def mark(m):
    t.write('\n===== restore2 {} {} =====\n'.format(time.strftime('%F %T'), m)); t.flush()
    print('##', time.strftime('%T'), m, flush=True)
def rd(dur):
    out = ''; end = time.time() + dur
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
    return out
def expect(pats, timeout, seed=''):
    out = seed; end = time.time() + timeout
    for p in pats:
        if re.search(p, out):
            return p, out
    while time.time() < end:
        b = s.read(4096)
        if b:
            x = b.decode(errors='replace'); out += x; t.write(x); t.flush()
            for p in pats:
                if re.search(p, out):
                    return p, out
    return None, out
def w(b):
    s.write(b); s.flush()
def cmd(c, timeout=60, allow_err=False):
    w(c.encode() + b'\r'); p, o = expect([r'\n[^\n]*#\s*$'], timeout)
    print('=== {} ===\n{}\n=== END ==='.format(c, o), flush=True)
    if p is None:
        mark('no prompt after {!r}'.format(c)); raise SystemExit(20)
    if re.search(r'^\s*% ', o, re.M) and not allow_err:
        mark('CLI error after {!r}: STOP'.format(c)); raise SystemExit(21)
    return o
def login():
    w(b'\r'); p, o = expect([r'login:\s*$', r'#\s*$', r'>\s*$', r'new password:\s*$'], 10)
    if p and 'new password' in p:
        mark('forced password dialog at login: STOP'); return False
    if p and 'login' in p:
        w(b'manager\r'); expect([r'Password:\s*$'], 10)
        w(PW.encode() + b'\r'); p, o = expect([r'#\s*$', r'>\s*$', r'Login incorrect', r'new password:\s*$'], 25)
        if p and 'new password' in p:
            mark('forced password dialog after login: STOP'); w(b'\x03'); return False
        if p == r'>\s*$':
            w(b'enable\r'); p, o = expect([r'#\s*$'], 10)
    return bool(p and '#' in p)
def escape_to_boot(tag):
    global state
    mark('ESCAPE [{}]'.format(tag)); o = rd(1.0)
    for _ in range(4):
        tail = o[-600:]
        if 'Quit and continue booting' in tail and tail.rstrip().endswith('==>'):
            w(b'9'); state = 'booting'; return
        if 'Return to previous menu' in tail and tail.rstrip().endswith('==>'):
            w(b'0'); o = rd(3.0); continue
        w(b'\x03'); o = rd(3.0)
    state = 'stuck'

PRE = '/home/terrenceb/claude/device-testing/x230v2-28GS/bootloader-6.2.40/5700.2001/work/u0-prerun-running-config.txt'
def norm(txt):
    out = []
    for l in txt.replace('\r', '').split('\n'):
        l = l.rstrip()
        if not l or l.startswith('show running-config') or re.match(r'^\S*#\s*$', l):
            continue
        l = re.sub(r'(username manager privilege 15 password 8 )\S+', r'\1<hash>', l)
        out.append(l)
    return out
def login_any(pw=None):
    pw = pw or PW
    w(b'\r'); p, o = expect([r'login:\s*$', r'awplus#\s*$', r'>\s*$', r'\]#\s*$', r'==>\s*$', r'new password:\s*$'], 10)
    if p and (']#' in p or '==>' in p or 'new password' in p):
        mark('unexpected console state {!r}: STOP'.format(p)); return False
    if p and 'login' in p:
        w(b'manager\r'); expect([r'Password:\s*$'], 10)
        w(pw.encode() + b'\r'); p, o = expect([r'#\s*$', r'>\s*$', r'Login incorrect', r'new password:\s*$'], 25)
        if p and 'Login incorrect' in p:
            mark('Login incorrect with the expected credential: STOP (no further attempts)'); return False
        if p and 'new password' in p:
            mark('forced dialog: STOP'); w(b'\x03'); return False
        if p == r'>\s*$':
            w(b'enable\r'); p, o = expect([r'#\s*$'], 10)
    return bool(p and '#' in p)
rc = 0
try:
    if not login_any(PW_TMP): rc = 2; sys.exit(rc)
    state = 'cli'
    cmd('terminal length 0')
    cmd('configure terminal'); cmd('username manager privilege 15 password friend'); cmd('end')
    now = time.strftime('%H:%M:%S %d %b %Y')
    cmd('clock set ' + now)
    cmd('show clock')
    cmd('configure terminal')
    for c in ('no ssh server allow-users manager', 'no service ssh', 'no loop-protection loop-detect',
              'interface vlan1', 'no ip dhcp-client vendor-identifying-class',
              'no ip dhcp-client request vendor-identifying-specific', 'no ipv6 address dhcp',
              'no ipv6 enable', 'no ip address dhcp', 'ip address 10.38.215.74/27', 'end'):
        cmd(c, allow_err=True)
    rd(3.0)
    cmd('ping 10.38.215.65 repeat 5', timeout=30, allow_err=True)
    o = cmd('copy tftp://10.38.215.65/x230v2_28GS-tb470.rel flash:/x230v2_28GS-tb470.rel', timeout=600, allow_err=True)
    if 'Successful operation' not in o:
        cmd('ping 10.38.215.65 repeat 5', timeout=30, allow_err=True)
        o = cmd('copy tftp://10.38.215.65/x230v2_28GS-tb470.rel flash:/x230v2_28GS-tb470.rel', timeout=600, allow_err=True)
    if 'Successful operation' not in o:
        mark('copy failed twice: STOP (unit runs x230-tb470.rel from TFTP; flash has backuprelease.rel)'); rc = 5; sys.exit(rc)
    cmd('configure terminal')
    for c in ('boot system flash:/x230v2_28GS-tb470.rel', 'no boot system backup', 'interface vlan1', 'no ip address', 'end'):
        cmd(c, allow_err=(c == 'no boot system backup'))
    cmd('delete force flash:/backuprelease.rel', allow_err=True)
    cmd('delete force flash:/TestCase_tear_down.cfg', allow_err=True)
    cmd('dir', allow_err=True)
    run = cmd('show running-config')
    a, b = norm(open(PRE).read()), norm(run)
    import difflib
    d = [l for l in difflib.unified_diff(a, b, 'prerun', 'now', lineterm='', n=0)]
    print('=== DIFF prerun vs now (username hash normalised) ===\n' + ('\n'.join(d) if d else '(identical)') + '\n=== END ===', flush=True)
    if d:
        mark('running-config differs from pre-run: STOP before writing default.cfg'); rc = 6; sys.exit(rc)
    cmd('copy running-config flash:/default.cfg')
    cmd('configure terminal'); cmd('boot config-file flash:/default.cfg'); cmd('end')
    cmd('show boot'); cmd('dir')
    mark('plain reload: expect TFTP default (forced banner)')
    w(b'reload\r'); p, o = expect([r'y/n'], 20)
    if not p: rc = 7; sys.exit(rc)
    w(b'y\r'); state = 'rebooting'
    p, o = expect([r'login:\s*$'], 900)
    i = o.find('Bootloader'); j = o.find('Booting')
    print('=== PROOF BOOT (verbatim) ===\n' + (o[i:j + 8] if i >= 0 and j > i else o[-2500:]) + '\n=== END ===', flush=True)
    mark('proof boot: login={} forced_banner={} tftp={} factory_cfg={}'.format(bool(p), 'non-standard location' in o, 'Loading tftp' in o, 'loading factory defaults' in o))
    if not p: rc = 8; state = 'unknown'; sys.exit(rc)
    state = 'login'
    if not login_any(): rc = 9; sys.exit(rc)
    state = 'cli'
    cmd('terminal length 0')
    for c in ('show system', 'show boot', 'show clock', 'dir', 'show license brief'):
        cmd(c, allow_err=True)
    run = cmd('show running-config')
    d = [l for l in difflib.unified_diff(norm(open(PRE).read()), norm(run), 'prerun', 'now', lineterm='', n=0)]
    print('=== FINAL DIFF prerun vs now ===\n' + ('\n'.join(d) if d else '(identical)') + '\n=== END ===', flush=True)
    mark('exit (log out)')
    w(b'exit\r'); expect([r'login:\s*$'], 15); state = 'loggedout'
finally:
    mark('close (state={}, rc={})'.format(state, rc))
    t.close(); s.close()
    open('/tmp/x230/run2/restore2.done', 'w').write('state={} rc={}\n'.format(state, rc))
